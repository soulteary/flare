var csrf = regexp.MustCompile(`(<input[^>]*name=(?:"csrf-token"|'csrf-token'|csrf-token)[^>]*value=")[^"]*(")`)
func normalize(b []byte)[]byte{return csrf.ReplaceAll(b,[]byte(`${1}CSRF_TOKEN${2}`))}
func digest(b []byte)string{h:=sha256.Sum256(b);return hex.EncodeToString(h[:])}
func emit(result *Result) {

  p.SetCookieCount = len(p.Headers.Values("Set-Cookie"))
  headers := http.Header{}
  for _, key := range []string{"Content-Type", "Location", "Cache-Control", "Referrer-Policy"} { if v:=p.Headers.Values(key);len(v)>0{headers[key]=v} }
  p.Headers=headers
  p.Body=csrf.ReplaceAllStringFunc(p.Body,func(s string)string{parts:=csrf.FindStringSubmatch(s);return parts[1]+strings.Repeat("X",len(s)-len(parts[1])-len(parts[2]))+parts[2]})
  p.Hash=digest([]byte(p.Body))
 }
 json.NewEncoder(os.Stdout).Encode(result)
}
